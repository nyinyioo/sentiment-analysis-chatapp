// Removes the contents of the given DOM element (equivalent to elem.innerHTML = '' but faster)
function emptyDOM (elem){
    while (elem.firstChild) elem.removeChild(elem.firstChild);
}

// Creates a DOM element from the given HTML string
function createDOM (htmlString){
    let template = document.createElement('template');
    template.innerHTML = htmlString.trim();
    return template.content.firstChild;
}

//create classes named Lobbyview, ChatView, and ProfileView
class LobbyView{
    //will create the DOM inside constructor and assign to elem
    constructor(lobby){
        this.lobby = lobby;
        this.elem = createDOM(`<div class = "content">
        <ul class = "room-list">
          <li><a href="#/chat/room-1">Chat Room 1</a></li>
          <li><a href="#/chat/room-2">Chat Room 2</a></li>
          <li><a href="#/chat/room-3">Chat Room 3</a></li>
        </ul>
        <br>
        <br>
        <div class="page-control">
          <input type="text" placeholder="Room Title">
          <button>Create Room</button>
        </div>
        </div>`);

        //store references to descendents elems for lobby DOM
        this.listElem = this.elem.querySelector('.room-list');
        this.inputElem = this.elem.querySelector('input[type="text"]');
        this.buttonElem = this.elem.querySelector('button');

        this.redrawList();
        
        var roomNum = 4;    
        this.buttonElem.addEventListener('click', () => {
            roomNum++;
            const roomName = this.inputElem.value;
            this.lobby.addRoom(roomNum, roomName);
            this.inputElem.value = ''; // Clear the text value
          });

        this.lobby.onNewRoom = (room) => {
            const listItem = document.createElement('li');
            listItem.innerHTML = "<a href=#/chat/room-" + room.id + ">"+ room.name +"</a>";
            this.listElem.appendChild(listItem);
        };

    }

    redrawList() {
        this.listElem.innerHTML = '';

        for (const roomId in this.lobby.rooms) {
            const room = this.lobby.rooms[roomId];
            const listItem = document.createElement('li');
            listItem.innerHTML = "<a href=#/chat/room-" + roomId + ">"+room.name+"</a>";
            this.listElem.appendChild(listItem);
          }
    }
}

class Room {
    //constructs the room object with default values for image and messages
    constructor(id, name, image = "assets/everyone-icon.png", messages = []) {
        this.id = id;
        this.name = name;
        this.image = image;
        this.messages = messages;
    }

    //new method: 
    addMessage(username, text){
        //if given text is empty string, whitespaces ==> return
        let message = {username, text};
        if (text.trim().length === 0){
            return;
        }
        else{
            //create new object and push to array messages.
            this.messages.push(message);
        }

        if (this.onNewMessage) {
            this.onNewMessage(message);
        }
    }
}

class Lobby {
    //constructs 4 rooms with id's
    constructor() {
        this.rooms = {
        '1': new Room('1', 'Room 1'),
        '2': new Room('2', 'Room 2'),
        '3': new Room('3', 'Room 3'),
        '4': new Room('4', 'Room 4')
      };
    }

    //getRoom by id method
    getRoom(roomId){
        if (roomId in this.rooms){
            return this.rooms[roomId];
        } 
        return null;
    }

    //adds a room to the room array
    addRoom(id, name, image, messages) {
        const newRoom = new Room(id, name, image, messages);
        this.rooms[id] = newRoom;
        if(this.onNewRoom) {
            this.onNewRoom(newRoom);
        }
    }
}

//create func main 
function main(){

    //lobby variable in main
    var lobby = new Lobby();

    //instantiate classes inside main
    const lobbyView = new LobbyView(lobby);
    const chatView = new ChatView();
    const profileView = new ProfileView();

    //create a func renderRoute inside main func
    function renderRoute(){
        //get the path and pageview to erase/populate content
        const hash = window.location.hash;
        //Case1: Chat Page
        if (hash.startsWith('#/chat/')){  
            //clear content of page-view
            const roomId = hash.split('-')[1];
            const room = lobby.getRoom(roomId);
            if (room) {
                chatView.setRoom(room);
            }
            emptyDOM(document.querySelector('#page-view'));
            //repopulate the contents of page-view from createDom
            document.querySelector('#page-view').appendChild(chatView.elem);
        }
        //Case2: Profile Page
        else if (hash == '#/profile'){
            //empty contents of page-view
            emptyDOM(document.querySelector('#page-view'));
            //repopulate contents of page-view
            document.querySelector('#page-view').appendChild(profileView.elem);
        }
        //Case3: Lobby Page
        else {
            //empty content 
            emptyDOM(document.querySelector('#page-view'));
            //repopulate
            document.querySelector('#page-view').appendChild(lobbyView.elem);
        }
    }
    //popstate event when URL changes
    window.addEventListener('popstate', renderRoute);
    //call renderRoute
    renderRoute();

    //Testing export local variables
    cpen322.export(arguments.callee, {renderRoute, lobbyView, chatView, profileView, lobby});
}

//load event for window object
window.addEventListener('load', main);
